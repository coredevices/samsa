cmake_minimum_required(VERSION 3.16)
project(samsa VERSION 0.1.0 LANGUAGES C)
option(SAMSA_BUILD_TESTS "Build unit tests" ON)
option(SAMSA_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(SAMSA_BUILD_BENCH "Build throughput benchmark" OFF)
option(SAMSA_BUILD_FUZZ "Build libFuzzer target (Clang)" OFF)
add_library(samsa src/samsa.c src/decode16.c src/decode32.c src/alu.c src/mem.c)
target_include_directories(samsa PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_compile_features(samsa PUBLIC c_std_11)
set_target_properties(samsa PROPERTIES C_EXTENSIONS OFF)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(samsa PRIVATE -Wall -Wextra -Werror -ffreestanding)
endif()
if(SAMSA_SANITIZE)
  target_compile_options(samsa PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(samsa PUBLIC -fsanitize=address,undefined)
endif()
if(SAMSA_BUILD_TESTS)
  enable_testing()
  file(GLOB SAMSA_TESTS CONFIGURE_DEPENDS tests/test_*.c)
  foreach(source IN LISTS SAMSA_TESTS)
    get_filename_component(name ${source} NAME_WE)
    add_executable(${name} ${source})
    target_link_libraries(${name} PRIVATE samsa)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
    target_compile_definitions(${name} PRIVATE SAMSA_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME ${name} COMMAND ${name})
  endforeach()
endif()
if(SAMSA_BUILD_BENCH)
  add_executable(samsa_bench tools/bench.c)
  target_link_libraries(samsa_bench PRIVATE samsa)
endif()
if(SAMSA_BUILD_FUZZ)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang")
    message(FATAL_ERROR "SAMSA_BUILD_FUZZ requires Clang")
  endif()
  target_link_options(samsa INTERFACE -fsanitize=address,undefined)
  target_compile_options(samsa PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
  add_executable(samsa_fuzz fuzz/fuzz.c)
  target_link_libraries(samsa_fuzz PRIVATE samsa)
  target_compile_options(samsa_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
  target_link_options(samsa_fuzz PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
include(GNUInstallDirs)
install(TARGETS samsa EXPORT samsaTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/samsa DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT samsaTargets NAMESPACE samsa:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/samsa)
